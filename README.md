## استفاده

```php
use Tavaan\Engine\Mail\MailManager;
use Tavaan\Engine\Mail\Message;
use Tavaan\Engine\Mail\Mailable;

// مستقیم
$m = new MailManager(['driver' => 'array']);
$m->send((new Message())->to('u@example.com')->subject('Test')->html('<p>Hi</p>'));

// با Mailable
class WelcomeMail extends Mailable {
    public function __construct(private readonly string $email) {}
    public function build(): self {
        return $this->subject('Welcome')->view('emails.welcome')
            ->to($this->email)->from('noreply@example.com')
            ->attach('/path/file.pdf');
    }
}
$m->to('user@example.com')->send(new WelcomeMail('user@example.com'));

// fluent
$m->to('u@example.com')->cc('admin@example.com')->send(new WelcomeMail('u@example.com'));

// Raw
$m->to('u@example.com')->raw('Plain text body', function (Message $m) {
    $m->subject('Subject');
});

// Fake (برای تست)
$m->fake();
$m->send((new Message())->to('u@example.com'));
$m->assertSent(WelcomeMail::class);
$m->sentMessages();  // list<Message>
```
